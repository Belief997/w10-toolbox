# 自制应援效果

设备只收到枚举。下面是编辑页预览的实现：文字在 `pages/cheer-edit/index.js`，背景在 `utils/cheer-light.js`。预览 24fps，运动按 466×466 画布定义，再乘 `预览像素 / 466`。

文字画在背景之后，不参与背景的缩放、遮罩、模糊、旋转。彩虹只改文字颜色。

## 时钟

```js
const RAINBOW_SPEED = 295; // 466 画布 px/s
const ROLL_SPEED = 215;    // 466 画布 px/s
const PREVIEW_FPS = 24;

this._textPhase += 1;
const step = css / 466 / PREVIEW_FPS;
if (scrollMode === 'scroll') this._scrollOffset += (css / 4) / PREVIEW_FPS;
else if (scrollMode === 'rainbow') this._scrollOffset += RAINBOW_SPEED * step;
else if (scrollMode === 'multiline') this._scrollOffset += ROLL_SPEED * step;
if (bgLightEffect !== 'none') this._lightPhase += (2 / 3) / PREVIEW_FPS;
this._timer = setTimeout(tick, 1000 / PREVIEW_FPS);
```

`phase` 的 1 个单位等于 1.5 秒。24fps 下每帧 `+1/36`。

```js
function effectSeconds(phase, speed) {
  const t = (phase || 0) * Math.max(0.25, speed || 1);
  return t * 1.5;
}
```

公共绘制：加粗，垂直居中，行距 1.35 倍字号。按 `\n` 拆行，去空行，最多 6 行。

```js
function cheerLines(text) {
  return String(text || '').split(/\n/).map((s) => s.trim()).filter(Boolean).slice(0, 6);
}

const scaledFont = fontSize * size / 466;
ctx.font = `bold ${scaledFont}px sans-serif`;
ctx.textBaseline = 'middle';
const lines = cheerLines(text);
const rowH = scaledFont * 1.35;
const stackStart = lines.length <= 1 ? size / 2 : size / 2 - ((lines.length - 1) * rowH) / 2;
const drawText = (line, x, y) => {
  ctx.fillStyle = ink;
  ctx.fillText(line, x, y);
};
const drawCentered = (line, y) => drawText(line, (size - ctx.measureText(line).width) / 2, y);
```

## textFx

| 值 | 名称 |
|----|------|
| `static` | 静止 |
| `scroll` | 走马灯 |
| `multiline` | 滚动 |
| `blink` | 闪烁 |
| `rainbow` | 彩虹 |

### static 静止

每行水平居中。单行在画布垂直中心，多行整块垂直居中。未知的 `textFx` 也走这里。

```js
const drawStack = () => {
  lines.forEach((line, i) => drawCentered(line, stackStart + i * rowH));
};
// else
drawStack();
```

### scroll 走马灯

从右向左。单行速度是画布宽度的 1/4 每秒，466 画布上 116.5 px/s。多行第 i 行（从 0 计）速度倍率 `1 + i × 0.4`。循环间距是「文字宽度 + 画布宽度」。

```js
// 每帧
this._scrollOffset += (css / 4) / PREVIEW_FPS;

const drawMarquee = (line, y, speedMul) => {
  const tw = ctx.measureText(line).width;
  const span = tw + size;
  const offset = (this._scrollOffset * dpr * (speedMul || 1)) % span;
  const x = size - offset;
  drawText(line, x, y);
  if (x + tw < size) drawText(line, x + tw + span, y);
};

if (scrollMode === 'scroll' && lines.length > 1) {
  lines.forEach((line, i) => drawMarquee(line, stackStart + i * rowH, 1 + i * 0.4));
} else if (scrollMode === 'scroll') {
  drawMarquee(lines[0] || text, size / 2, 1);
}
```

### multiline 滚动

整块自下而上，每行水平居中。466 画布 215 px/s。整块离开顶部后从底部再进入。循环长度是「整块高度 + 画布高度」。

```js
// 每帧
this._scrollOffset += ROLL_SPEED * (css / 466 / PREVIEW_FPS);

const drawRoll = () => {
  const blockH = Math.max(rowH, lines.length * rowH);
  const span = blockH + size;
  const offset = ((this._scrollOffset || 0) * dpr) % span;
  const startY = size - offset + rowH / 2;
  lines.forEach((line, i) => drawCentered(line, startY + i * rowH));
};
```

### blink 闪烁

位置与静止相同。1 秒一个周期，透明度在 1 和 0 之间按余弦往复。`t = _textPhase / 24`。

```js
const sec = (this._textPhase || 0) / PREVIEW_FPS;
ctx.save();
ctx.globalAlpha = 0.5 + 0.5 * Math.cos(Math.PI * 2 * sec);
drawStack();
ctx.restore();
```

### rainbow 彩虹

各行同速横滚，倍率都是 1。466 画布 295 px/s。不使用 `textColor`。整段同一颜色，饱和度 1、明度 0.5，色相每秒 +36°，10 秒转一圈。

```js
// 每帧
this._scrollOffset += RAINBOW_SPEED * (css / 466 / PREVIEW_FPS);

lines.forEach((line, i) => drawMarquee(line, stackStart + i * rowH, 1));

_inkColor() {
  if (this.data.scrollMode !== 'rainbow') return this.data.fontColor;
  const hue = (((this._textPhase || 0) / PREVIEW_FPS) * 36) % 360;
  const s = 1;
  const l = 0.5;
  const a = s * Math.min(l, 1 - l);
  const f = (n) => {
    const k = (n + hue / 30) % 12;
    const c = l - a * Math.max(Math.min(k - 3, 9 - k, 1), -1);
    return Math.round(255 * c).toString(16).padStart(2, '0');
  };
  return `#${f(0)}${f(8)}${f(4)}`;
}
```

`hue / 30` 是 HSL 转 RGB 的扇区计算，与帧率无关。

## bgFx

先铺黑底，再画纯色或图片。图片按短边居中裁成正方形。不认识的值按静止全亮画。

```js
function paintCover(ctx, size, opts, alpha) {
  if (alpha <= 0) return;
  ctx.save();
  ctx.globalAlpha = Math.max(0, Math.min(1, alpha));
  if (opts.bgImage) drawImageCover(ctx, opts.bgImage, size);
  else {
    ctx.fillStyle = opts.bgColor || '#000000';
    ctx.fillRect(0, 0, size, size);
  }
  ctx.restore();
}

function drawCheerBackground(ctx, size, opts) {
  const effect = opts.effect;
  const sec = effectSeconds(opts.phase, opts.speed);
  ctx.fillStyle = '#000000';
  ctx.fillRect(0, 0, size, size);
  // 按 effect 分支
}
```

| 值 | 名称 | 周期 |
|----|------|------|
| `none` | 静止 | 不动 |
| `breathe` | 呼吸 | 2 秒 |
| `pulse` | 爆闪 | 0.4 秒 |
| `open` | 开幕 | 2.5 秒 |
| `blur` | 模糊 | 1.5 秒 |
| `rotate` | 旋转 | 2 秒 |

发出前改写：`step` → `open`，`shrink` → `blur`，`zoom` → `rotate`。

### none 静止

```js
if (!effect || effect === 'none') {
  paintCover(ctx, size, opts, 1);
  return;
}
```

### breathe 呼吸

2 秒一轮。第 1 秒不透明度从 1 线性到 0，第 2 秒从 0 线性到 1。

```js
if (effect === 'breathe') {
  const u = wrap(sec, 2);
  paintCover(ctx, size, opts, u < 1 ? 1 - u : u - 1);
  return;
}
```

### pulse 爆闪

0.4 秒一轮。前 0.2 秒画背景，后 0.2 秒留黑底。

```js
if (effect === 'pulse') {
  if (wrap(sec, 0.4) < 0.2) paintCover(ctx, size, opts, 1);
  return;
}
```

### open 开幕

2.5 秒一轮。左右黑遮罩从中线向外打开。`shift` 为 0.25 时，466 画布上每侧移开 116.5 px。

| 时间 | shift | 缩放 |
|------|-------|------|
| 0–0.5 秒 | 0 → 0.25 | 150% → 110% |
| 0.5–1 秒 | 0.25 | 110% |
| 1–1.5 秒 | 0.25 → 0.5 | 110% → 100% |
| 1.5–2.5 秒 | 0.5 | 100% |

```js
function drawOpen(ctx, size, opts, sec) {
  const u = wrap(sec, 2.5);
  let shift = 0.5;
  let scale = 1;
  if (u < 0.5) {
    const t = u / 0.5;
    shift = lerp(0, 0.25, t);
    scale = lerp(1.5, 1.1, t);
  } else if (u < 1) {
    shift = 0.25;
    scale = 1.1;
  } else if (u < 1.5) {
    const t = (u - 1) / 0.5;
    shift = lerp(0.25, 0.5, t);
    scale = lerp(1.1, 1, t);
  }
  paintCoverScaled(ctx, size, opts, scale);
  drawCurtains(ctx, size, shift * size);
}

function drawCurtains(ctx, size, shift) {
  const edge = size / 2 - shift;
  if (edge <= 0) return;
  ctx.save();
  ctx.fillStyle = '#000000';
  ctx.fillRect(0, 0, edge, size);
  ctx.fillRect(size - edge, 0, edge, size);
  ctx.restore();
}
```

### blur 模糊

1.5 秒一轮。0–0.5 秒缩放 150% → 100%，高斯模糊 10 → 0（466 画布像素，再乘 `预览像素 / 466`）。之后保持清晰。不支持 `ctx.filter` 时只做缩放。

```js
function drawBlur(ctx, size, opts, sec) {
  const u = wrap(sec, 1.5);
  let scale = 1;
  let blur = 0;
  if (u < 0.5) {
    const t = u / 0.5;
    scale = lerp(1.5, 1, t);
    blur = lerp(10, 0, t) * size / 466;
  }
  ctx.save();
  if (blur > 0.2) {
    try { ctx.filter = `blur(${blur}px)`; } catch (err) {}
  }
  paintCoverScaled(ctx, size, opts, scale);
  try { ctx.filter = 'none'; } catch (err) {}
  ctx.restore();
}
```

### rotate 旋转

2 秒一轮。第 1 秒转到正位，第 2 秒保持。四个同心圆从外到内裁切后各自旋转。

| 环 | 466 直径 | 半径 / 边长 | 角度 |
|----|----------|-------------|------|
| 外 | 466 | 0.5 | -90° → 0° |
| 2 | 349.5 | 0.375 | 90° → 0° |
| 3 | 233 | 0.25 | -90° → 0° |
| 内 | 116.5 | 0.125 | 90° → 0° |

```js
function drawRotate(ctx, size, opts, sec) {
  const u = wrap(sec, 2);
  const t = u < 1 ? u : 1;
  const odd = (90 * (1 - t)) * Math.PI / 180;
  const even = (-90 * (1 - t)) * Math.PI / 180;
  const rings = [
    { radius: size * 0.5, angle: even },
    { radius: size * 0.375, angle: odd },
    { radius: size * 0.25, angle: even },
    { radius: size * 0.125, angle: odd }
  ];
  rings.forEach((ring) => {
    ctx.save();
    ctx.beginPath();
    ctx.arc(size / 2, size / 2, ring.radius, 0, Math.PI * 2);
    ctx.clip();
    ctx.translate(size / 2, size / 2);
    ctx.rotate(ring.angle);
    ctx.translate(-size / 2, -size / 2);
    paintCover(ctx, size, opts, 1);
    ctx.restore();
  });
}
```
